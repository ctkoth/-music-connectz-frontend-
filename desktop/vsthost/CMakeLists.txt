# mcz-vsthost: the desktop app's VST3 host. Built in CI (windows-exe.yml) and testable on Linux.
#   cmake -B build -DVST3_SDK=/path/to/vst3sdk && cmake --build build --config Release
cmake_minimum_required(VERSION 3.25)
project(mcz_vsthost CXX)
set(CMAKE_CXX_STANDARD 17)
set(VST3_SDK "" CACHE PATH "Steinberg VST3 SDK checkout (MIT licence)")
if(NOT VST3_SDK)
  message(FATAL_ERROR "Set -DVST3_SDK=<path to vst3sdk>")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory(${VST3_SDK} vst3sdk EXCLUDE_FROM_ALL)
# The SDK's own hosting library (module loading, PlugProvider, process data, parameter changes, IIDs).
set(HOST ${VST3_SDK}/public.sdk/source/vst/hosting)
set(SRC mcz_vsthost.cpp ${HOST}/module.cpp ${HOST}/plugprovider.cpp)
if(WIN32)
  list(APPEND SRC ${HOST}/module_win32.cpp)
elseif(APPLE)
  list(APPEND SRC ${HOST}/module_mac.mm)
else()
  list(APPEND SRC ${HOST}/module_linux.cpp)
endif()
add_executable(mcz-vsthost ${SRC})
target_include_directories(mcz-vsthost PRIVATE ${VST3_SDK})
target_link_libraries(mcz-vsthost PRIVATE sdk_hosting sdk_common pluginterfaces base)
if(UNIX AND NOT APPLE)
  target_link_libraries(mcz-vsthost PRIVATE dl stdc++fs)
endif()
if(MSVC)
  target_compile_options(mcz-vsthost PRIVATE /utf-8)
  set_property(TARGET mcz-vsthost PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
